local function formatNumber(n)
	if not n then return "0" end
	return tostring(math.floor(n)):reverse():gsub("(%d%d%d)", "%1,"):reverse():gsub("^,", "")
end

local function fadeIn(frame, onDone)
	frame.BackgroundTransparency = 1
	frame.Visible = true
	for _, c in ipairs(frame:GetDescendants()) do
		if c:IsA("TextLabel") or c:IsA("TextButton") then
			pcall(function() c.TextTransparency = 1 end)
		elseif c:IsA("ImageLabel") or c:IsA("ImageButton") then
			pcall(function() c.ImageTransparency = 1 end)
		end
	end
	local t = TweenService:Create(frame, TweenInfo.new(ANIM, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {BackgroundTransparency = 0})
	for _, c in ipairs(frame:GetDescendants()) do
		if c:IsA("TextLabel") or c:IsA("TextButton") then
			pcall(function() TweenService:Create(c, TweenInfo.new(ANIM), {TextTransparency = 0}):Play() end)
		elseif c:IsA("ImageLabel") or c:IsA("ImageButton") then
			pcall(function() TweenService:Create(c, TweenInfo.new(ANIM), {ImageTransparency = 0}):Play() end)
		end
	end
	t:Play()
	t.Completed:Connect(function() if onDone then onDone() end end)
end

local function fadeOut(frame, onDone)
	local t = TweenService:Create(frame, TweenInfo.new(ANIM, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {BackgroundTransparency = 1})
	for _, c in ipairs(frame:GetDescendants()) do
		if c:IsA("TextLabel") or c:IsA("TextButton") then
			pcall(function() TweenService:Create(c, TweenInfo.new(ANIM), {TextTransparency = 1}):Play() end)
		elseif c:IsA("ImageLabel") or c:IsA("ImageButton") then
			pcall(function() TweenService:Create(c, TweenInfo.new(ANIM), {ImageTransparency = 1}):Play() end)
		end
	end
	t:Play()
	t.Completed:Connect(function() frame.Visible = false; if onDone then onDone() end end)
end

local function makeDraggable(frame, handle)
	local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
	local target = handle or frame
	target.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true; dragStart = input.Position; startPos = frame.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false; dragInput = nil end
			end)
		end
	end)
	target.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local d = input.Position - dragStart
			frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
end

local function hideMainGameUI()
	hiddenUIElements = {}; isUIRestoreLocked = true
	for _, g in ipairs(playerGui:GetChildren()) do
		if g:IsA("ScreenGui") and g.Name ~= "GiftInterface" then
			hiddenUIElements[g] = g.Enabled; g.Enabled = false
		end
	end
end

local function restoreMainGameUI()
	if not isUIRestoreLocked then return end
	for g, v in pairs(hiddenUIElements) do if g and g.Parent then g.Enabled = v end end
	hiddenUIElements = {}; isUIRestoreLocked = false
end

		recipientLabel.Text = SelectedRecipient.displayName .. " (@" .. SelectedRecipient.username .. ")"

local function renderFriends(filter)
	filter = (filter or ""):lower()
	for _, c in ipairs(friendsScroll:GetChildren()) do
		if c:IsA("TextButton") then c:Destroy() end
	end
	friendRowsByUser = {}
	for _, f in ipairs(allFriends) do
		if filter == "" or f.username:lower():find(filter,1,true) or f.displayName:lower():find(filter,1,true) then
			createFriendRow(f.displayName, f.username, f.userId, f.thumb)
		end
	end
	task.wait()
	friendsScroll.CanvasSize = UDim2.new(0, 0, 0, friendsLayout.AbsoluteContentSize.Y + 8)
	local empty = #allFriends == 0
	friendsStatusLabel.Visible = empty
	if empty then friendsStatusLabel.Text = "No friends found" end
end

local function loadFriends()
	if friendsLoaded then return end
	friendsLoaded = true
	task.spawn(function()
		local ok, pages = pcall(function() return Players:GetFriendsAsync(player.UserId) end)
		if not ok or not pages then friendsStatusLabel.Text = "Could not load friends"; return end
		local count = 0
		while true do
			for _, item in ipairs(pages:GetCurrentPage()) do
				local thumb = ""
				pcall(function()
					thumb = Players:GetUserThumbnailAsync(item.Id, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
				end)
				table.insert(allFriends, {userId=item.Id, username=item.Username, displayName=item.DisplayName or item.Username, thumb=thumb})
				count += 1
				if count % 8 == 0 then renderFriends(searchBox.Text) end
			end
			if pages.IsFinished then break end
			local ok2 = pcall(function() pages:AdvanceToNextPageAsync() end)
			if not ok2 then break end
		end
		renderFriends(searchBox.Text)
	end)
end

searchBox:GetPropertyChangedSignal("Text"):Connect(function()
	if currentTab == "friends" then renderFriends(searchBox.Text) end
end)

-- HTTP HELPER

local function httpRequest(url)
	local req = (syn and syn.request) or (http and http.request) or http_request or request
	if not req then return nil end
	local ok, res = pcall(req, {Url=url, Method="GET"})
	if ok and res and (res.StatusCode==200 or res.Status==200) and res.Body then return res.Body end
	return nil
end

local _joinYearCache, _mutualCache, _localFriendIds = {}, {}, nil

local function getLocalFriendIds()
	if _localFriendIds then return _localFriendIds end
	_localFriendIds = {}
	local ok, pages = pcall(function() return Players:GetFriendsAsync(player.UserId) end)
	if not ok or not pages then return _localFriendIds end
	while true do
		for _, item in ipairs(pages:GetCurrentPage()) do _localFriendIds[item.Id] = true end
		if pages.IsFinished then break end
		if not pcall(function() pages:AdvanceToNextPageAsync() end) then break end
	end
	return _localFriendIds
end

local function fetchJoinYear(userId, cb)
	task.spawn(function()
		if _joinYearCache[userId] ~= nil then cb(_joinYearCache[userId]); return end
		local body = httpRequest("https://users.roblox.com/v1/users/"..userId)
		local y = nil
		if body then local yr = body:match('"created"%s*:%s*"(%d%d%d%d)'); if yr then y = tonumber(yr) end end
		_joinYearCache[userId] = y or false; cb(y)
	end)
end

local function fetchMutualCount(userId, cb)
	task.spawn(function()
		if _mutualCache[userId] ~= nil then cb(_mutualCache[userId]); return end
		if userId == player.UserId then _mutualCache[userId]=0; cb(0); return end
		local localSet = getLocalFriendIds()
		local ok, pages = pcall(function() return Players:GetFriendsAsync(userId) end)
		if not ok or not pages then _mutualCache[userId]=0; cb(0); return end
		local count = 0
		while true do
			for _, item in ipairs(pages:GetCurrentPage()) do if localSet[item.Id] then count+=1 end end
			if pages.IsFinished then break end
			if not pcall(function() pages:AdvanceToNextPageAsync() end) then break end
		end
		_mutualCache[userId]=count; cb(count)
	end)
end

local pendingAmount = 0

local function stopLoading()
	if loadingConn then pcall(function() loadingConn:Disconnect() end); loadingConn = nil end
	isLoading = false
	sendBtn.Active = true
	sendBtn.BackgroundColor3 = Color3.fromRGB(0,120,255)
	clipping.Size = UDim2.new(0,0,1,0)
	fillBar.Size = UDim2.new(1,0,1,0)

local function startLoading()
	stopLoading()
	isLoading = true
	sendBtn.Active = false
	sendBtn.BackgroundColor3 = Color3.fromRGB(0,60,140)
	clipping.Size = UDim2.new(0,0,1,0)
	local speedVal = math.max(LoadSpeed, 0.1)
	task.wait(0.2)
	local startTime = tick()
	loadingConn = RunService.RenderStepped:Connect(function()
		local progress = math.clamp((tick()-startTime)/speedVal, 0, 1)
		clipping.Size = UDim2.new(progress, 0, 1, 0)
		fillBar.Size = UDim2.new(progress == 0 and 0 or 1/progress, 0, 1, 0)
		if progress >= 1 then
			loadingConn:Disconnect(); loadingConn = nil
			isLoading = false
			sendBtn.Active = true
			sendBtn.BackgroundColor3 = Color3.fromRGB(0,120,255)
			clipping.Size = UDim2.new(1,0,1,0)
			fillBar.Size = UDim2.new(1,0,1,0)
			btnLabel.Text = "Send"
		end
	end)
end

local currentSearchId = 0

local function doUsernameSearch(username)
	currentSearchId += 1
	local thisId = currentSearchId
	if not username or username == "" then
		stopSpinner(); searchProfilePic.Image = ""
		displayNameLabel.Text = ""; usernameLabel.Text = ""
		lookupCard.Visible = false
		if currentTab == "username" then lookupHint.Visible = true end
		return
	end
	lookupHint.Visible = false; lookupCard.Visible = false
	startSpinner()
	task.spawn(function()
		local ok, userId = pcall(function() return Players:GetUserIdFromNameAsync(username) end)
		if thisId ~= currentSearchId then return end
		stopSpinner()
		if ok and userId then
			local popSound = Instance.new("Sound")
			popSound.SoundId = ASSETS.PopSound; popSound.Volume = 0.5
			popSound.Parent = giftFrame; popSound:Play()
			game.Debris:AddItem(popSound, 1)
			local thumbOk, thumb = pcall(function()
				return Players:GetUserThumbnailAsync(userId, Enum.ThumbnailType.HeadShot, Enum.ThumbnailSize.Size150x150)
			end)
			if thisId ~= currentSearchId then return end
			local infoOk, info = pcall(function() return UserService:GetUserInfosByUserIdsAsync({userId}) end)
			local dn, un = username, username
			if infoOk and info and info[1] then dn = info[1].DisplayName; un = info[1].Username end
			if thisId ~= currentSearchId then return end
			searchProfilePic.Image = (thumbOk and thumb and thumb~="") and thumb or ""
			displayNameLabel.Text = dn; usernameLabel.Text = "@"..un
			lookupHint.Visible = false; lookupCard.Visible = true
			setRecipient(un, dn, userId)
		else
			if thisId ~= currentSearchId then return end
			searchProfilePic.Image = ""; displayNameLabel.Text = "Player not found"
			usernameLabel.Text = ""; lookupHint.Visible = false; lookupCard.Visible = true
		end
	end)
end

usernameBox:GetPropertyChangedSignal("Text"):Connect(function()
	if usernameBox.Text == "" then doUsernameSearch("") end
end)
usernameBox.FocusLost:Connect(function()
	if usernameBox.Text ~= "" then doUsernameSearch(usernameBox.Text) end
end)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then return end
	if input.KeyCode == Enum.KeyCode.Return and usernameBox:IsFocused() and usernameBox.Text ~= "" then
		doUsernameSearch(usernameBox.Text); usernameBox:ReleaseFocus()
	end
end)
